```ts
import { withStore } from '@ebda/database';

export interface StoreContactTarget {
  /** The store's public contact email, else its owner's account email; null when neither exists. */
  to: string | null;
  isDemo: boolean;
  slug: string;
}

/**
 * Where a shopper's storefront contact message goes (Ruling FU46-1): the address the store shows
 * on its contact page (`stores.email`), else the owner's account email. Read in the store's own
 * scope: the owner's `users` row is visible there (strict RLS policy on `users`, owner EXISTS
 * clause). Null when the store no longer exists or is soft-deleted.
 */
export async function storeContactTarget(storeId: string): Promise<StoreContactTarget | null> {
  const store = await withStore(storeId, tx =>
    tx.store.findUnique({
      where: { id: storeId },
      select: {
        email: true,
        isDemo: true,
        slug: true,
        deletedAt: true,
        owner: { select: { email: true } },
      },
    })
  );
  if (!store || store.deletedAt) return null;
  const to = store.email?.trim() || store.owner?.email?.trim() || null;
  return { to, isDemo: store.isDemo, slug: store.slug };
}
```

import { inngest } from '../client';

type ContactMessage = { name: string; email: string; subject: string; message: string };
export interface ContactStep {
  run: <T>(id: string, fn: () => Promise<T>) => Promise<T>;
}

/**
 * Sends a storefront contact-page message to the STORE it was written on (Ruling FU46-1):
 * the store's public contact email, else the owner's account email, else (no address at all)
 * the platform inbox labelled with the store. Before FU46-1 every store's messages went to
 * `config.smtpFromEmail` — Ebda's own sender — and no merchant ever saw one.
 *
 * #645 still holds: queued on Inngest (3 retries, dead-letter via alerts-failure-watch); a
 * send failure is rethrown so it retries. A store that is gone or turned into a demo store
 * between submit and send gets nothing — there is no one to deliver to, so no retry either.
 */
export async function handleContactFormSubmitted({
  event,
  step,
}: {
  event: { name: string; data: unknown };
  step: ContactStep;
}): Promise<void> {
  const { storeId, name, email, subject, message } = (event.data ?? {}) as Partial<
    ContactMessage & { storeId: string }
  >;
  if (!name || !email || !subject || !message) {
    logger.warn({ event: event.name }, 'contact/form.submitted fired without required fields');
    return;
  }
  const data: ContactMessage = { name, email, subject, message };

  if (!storeId) {
    // Enqueued before FU46-1 deployed: keep the destination it was enqueued for.
    await step.run('send-email', () => EmailService.sendContactForm(data));
    return;
  }

  await step.run('send-email', async () => {
    const target = await storeContactTarget(storeId);
    if (!target || target.isDemo) {
      logger.warn({ storeId }, '[CONTACT] store gone or demo; message dropped');
      return;
    }
    try {
      if (target.to) await EmailService.sendStoreContactMessage(storeId, target.to, data);
      else await EmailService.sendContactForm(data, { storeLabel: target.slug });
    } catch (err) {
      logger.error({ err, storeId, email: maskEmail(email) }, '[CONTACT] message email failed');
      throw err; // Inngest retries; alerts-failure-watch pages after the final one.
    }
  });
}

export const contactFormSend: InngestFunction.Any = inngest.createFunction(
  {
    id: 'contact-form-send',
    name: 'Send a storefront contact message to its store',
    retries: 3,
    triggers: [{ event: 'contact/form.submitted' }],
  },
  async ({ event, step }) =>
    handleContactFormSubmitted({ event, step: step as unknown as ContactStep })
);

Create `apps/api/src/lib/upload-sniff.ts`:

```ts
/**
 * Magic-byte detection for the image and font uploads (B0-46; the video twin is
 * lib/video-sniff.ts). The multipart mime and the filename are attacker-controlled; the
 * bytes decide what a file is and therefore its stored extension and Content-Type.
 */
export type ImageKind = 'image/jpeg' | 'image/png' | 'image/gif' | 'image/webp';
export const IMAGE_EXT: Record<ImageKind, string> = {
  'image/jpeg': '.jpg',
  'image/png': '.png',
  'image/gif': '.gif',
  'image/webp': '.webp',
};

const PNG_MAGIC = Buffer.from([0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a]);

export function sniffImage(buf: Buffer): ImageKind | null {
  if (buf.length >= 3 && buf[0] === 0xff && buf[1] === 0xd8 && buf[2] === 0xff) return 'image/jpeg';
  if (buf.length >= 8 && buf.subarray(0, 8).equals(PNG_MAGIC)) return 'image/png';
  if (buf.length >= 6) {
    const head = buf.subarray(0, 6).toString('latin1');
    if (head === 'GIF87a' || head === 'GIF89a') return 'image/gif';
  }
  if (
    buf.length >= 12 &&
    buf.subarray(0, 4).toString('latin1') === 'RIFF' &&
    buf.subarray(8, 12).toString('latin1') === 'WEBP'
  ) {
    return 'image/webp';
  }
  return null;
}

export type FontKind = 'font/woff2' | 'font/woff' | 'font/ttf' | 'font/otf';
export const FONT_EXT: Record<FontKind, string> = {
  'font/woff2': '.woff2',
  'font/woff': '.woff',
  'font/ttf': '.ttf',
  'font/otf': '.otf',
};

export function sniffFont(buf: Buffer): FontKind | null {
  if (buf.length < 4) return null;
  const tag = buf.subarray(0, 4).toString('latin1');
  if (tag === 'wOF2') return 'font/woff2';
  if (tag === 'wOFF') return 'font/woff';
  if (tag === 'OTTO') return 'font/otf';
  if (tag === 'true' || buf.readUInt32BE(0) === 0x00010000) return 'font/ttf';
  return null;
}

/** `photo.png` + `image/jpeg` → `photo.jpg`; an empty base becomes `fallback`. */
export function renameByKind(originalName: string, ext: string, fallback: string): string {
  return `${originalName.replace(/\.[^.]*$/, '') || fallback}${ext}`;
}
```

/**
 * Issue a form token: `<issuedAtMs>.<nonce>.<hmac>`. Stateless; single use is enforced at submit
 * by claiming the signature. The nonce makes two tokens issued in the same millisecond distinct:
 * without it they were identical and the second visitor's submit was refused as already used
 * (Ruling FU46-8).
 */
export function issueSignupFormToken(now: number = Date.now()): string {
  const ts = String(now);
  const nonce = crypto.randomBytes(9).toString('base64url');
  return `${ts}.${nonce}.${sign(`${ts}.${nonce}`)}`;
}

export type FormTokenVerdict = 'ok' | 'invalid' | 'too_fast' | 'expired';

const NONCE_RE = /^[A-Za-z0-9_-]{12}$/;

export function verifySignupFormToken(
  token: unknown,
  now: number = Date.now()
): { verdict: FormTokenVerdict; sig?: string } {
  if (typeof token !== 'string') return { verdict: 'invalid' };
  const parts = token.split('.');
  // 3 parts since FU46-8; 2 parts = a token issued before it, accepted until it ages out.
  if (parts.length !== 2 && parts.length !== 3) return { verdict: 'invalid' };
  const ts = parts[0];
  const nonce = parts.length === 3 ? parts[1] : null;
  const sig = parts[parts.length - 1];
  if (!/^\d{10,15}$/.test(ts)) return { verdict: 'invalid' };
  if (nonce !== null && !NONCE_RE.test(nonce)) return { verdict: 'invalid' };
  const expected = sign(nonce === null ? ts : `${ts}.${nonce}`);
  const a = Buffer.from(sig);
  const b = Buffer.from(expected);
  if (a.length !== b.length || !crypto.timingSafeEqual(a, b)) return { verdict: 'invalid' };
